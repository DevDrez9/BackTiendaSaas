import { AuthService } from '../src/auth/auth.service';
const users: any[] = [{ id: 1, email: 'a@b.com', nombre: 'Ana', activo: true, password: 'old' }];
let codes: any[] = []; let enviado = '';
const cr = {
  findFirst: async ({ where, orderBy }: any) => {
    let l = codes.filter(c => c.usuarioId === where.usuarioId);
    if ('usadoEn' in where) l = l.filter(c => c.usadoEn === null && c.expiraEn > new Date());
    return l.sort((a,b)=>b.createdAt-a.createdAt)[0] ?? null;
  },
  updateMany: async ({ where, data }: any) => { codes.filter(c=>c.usuarioId===where.usuarioId && c.usadoEn===null).forEach(c=>Object.assign(c,data)); return {}; },
  create: async ({ data }: any) => { codes.push({ id: codes.length+1, intentos: 0, usadoEn: null, createdAt: new Date(), ...data }); return {}; },
  update: async ({ where, data }: any) => { const c = codes.find(x=>x.id===where.id); c.intentos += 1; return c; },
};
const prisma: any = {
  usuario: { findUnique: async ({ where }: any) => users.find(u => u.email === where.email) ?? null, update: async ({ data }: any) => { users[0].password = data.password; return users[0]; } },
  codigoRecuperacion: cr,
  $transaction: async (ops: any[]) => Promise.all(ops),
};
const mail: any = { enviar: async (_: string, asunto: string) => { enviado = asunto; return true; }, plantilla: (t: string, h: string) => h };

  await t('reset bien', a.resetPassword({ email: 'a@b.com', codigo, password: 'nueva123' }));
